111double Fish::naturalMortalityRate(double temp) const{
112 double rate;
113 if (age > par.amax) return 1e20; // FIXME: use inf
114 else {
115 if (par.mortality_model == MortalityModel::Dankel22){
116 if (isMature) rate = par.mam[age];
117 else rate = par.mai[age];
118 return rate;
119 }
120 else if (par.mortality_model == MortalityModel::Bioenergetic){
121 //return fish::natural_mortality(length, temp, par.M0, par.gamma3, par.alpha3, par.Lref, par.Tref, par.cT);
122 // Note: Mortality due to spawning is not included here, it is treated separately in the reproduction part of the population update.
123 return (par.M0 +
124 par.alpha3 * pow(length / par.Lref, par.gamma3) +
125 par.alpha4*(par.alpha1*par.alpha1 - par.alpha1_ref*par.alpha1_ref) +
126 par.alpha5*(par.gsi - par.gsi_ref)
127 ) * pow(temp/par.Tref, par.cT);
128 }
129 else{
130 throw std::runtime_error("Invalid mortality model specified");
131 }
132 }
133}

171void Fish::grow(double tsb, double temp){
172 if (par.growth_model == GrowthModel::Dankel22){
173 // do nothing. age is incremented by population update
174 gsi_effective = par.gsi; // required if new fecundity model is used in combination with old growth model
175 }
176 else if (par.growth_model == GrowthModel::Bioenergetic){
177 double tsb_ano = tsb - par.tsbmean;
178 double temp_ano = temp - par.Tmean;
179
180 // This is generalized increment l2^y1y2 - l1^y1y2
181 double dl = fish::dl_power(tsb_ano, temp_ano, par.gamma1, par.gamma2, par.alpha1, par.alpha2, par.beta1, par.beta2);
182 double dl_pot = fish::dl_power(-par.tsbmean, temp_ano, par.gamma1, par.gamma2, par.alpha1, par.alpha2, par.beta1, par.beta2);
183
184 double lnew, lnew_pot;
185 if (isMature){
186 lnew = fish::length_adult(length, dl, par.gamma1, par.gamma2, par.gsi);
187 lnew_pot = fish::length_adult(length, dl_pot, par.gamma1, par.gamma2, par.gsi);
188 }
189 else{
190 lnew = fish::length_juvenile(length, dl, par.gamma1, par.gamma2);
191 lnew_pot = fish::length_juvenile(length, dl_pot, par.gamma1, par.gamma2);
192 }
193
194 // ------ This is linear increment, just for analysis --------
195 dl_real = lnew - length;
196 dl_potential = lnew_pot - length;
197 //cout << "tsb_ano = " << tsb << " / " << par.tsbmean << ", fac = " << dl_real << " / " << dl_potential << endl;
198 // -----------------------------------------------------------
199
200 gsi_effective = fish::gsi(lnew, length, dl, par.gamma1, par.gamma2, par.alpha1, par.alpha2);
201 set_length(lnew);
202 }
203 else{
204 throw std::runtime_error("Invalid growth model specified");
205 }
208}

247double Fish::produceRecruits(double ssb, double temp){
248 double temp_ano = temp - par.Tref;
249 if (par.recruitment_model == RecruitmentModel::BevertonHoltDirect){
250 double recruits = par.r0 * weight * 1 / (1 + ssb/par.Bhalf);
251 return recruits;
252 }
253 else if (par.recruitment_model == RecruitmentModel::RickerDirect){
254 double recruits = par.r0 * weight * exp(par.beta4*temp_ano) * pow(2, -ssb/par.Bhalf);
255 return recruits;
256 }
257 else if (par.recruitment_model == RecruitmentModel::BevertonHoltBioenergetic){
258 double eggs = fish::fecundity(weight, par.delta, gsi_effective); // Total eggs produced
259 double recruits = eggs * par.s0 * 1 / (1 + ssb/par.Bhalf); // offspring surviving density-dependent recruitment
260 return recruits;
261 }
262 else if (par.recruitment_model == RecruitmentModel::RickerBioenergetic){
263 double eggs = fish::fecundity(weight, par.delta, gsi_effective); // Total eggs produced
264 double recruits = eggs * par.s0 * exp(par.beta4*temp_ano) * pow(2, -ssb/par.Bhalf); // offspring surviving density-dependent recruitment
265 return recruits;
266 }
267 else{
268 throw std::runtime_error("Invalid recruitment model specified");
269 }
