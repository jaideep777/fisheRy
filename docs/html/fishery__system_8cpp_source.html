148double Fishery::calc_quota(double temp){
149 // FishingMortalityRef curve applies to the entire fishery, so we calculate quota from the effective fleet
150 auto& fl = fleet_effective;
151 double Fc = -log(1-harvest_prop);
152
153 // Calculate chi of the fleet to implement mortality
154 fl.chi = Fc/fl.FishingMortalityRef_avgl(200, 100);
155
156 // Calculate expected Catch with the given chi
157 // Note: Be careful to not modify the fishes vector here
158 double expected_catch = 0, to_sea_bed = 0;
159 // int count = 0;
160 // std::cout << "Start catch: min size limit = " << par.lmin << '\n';
161 for (const auto& f: pop.fishes) { // const auto& ensures we do not modify the fish
162 if (!f.isAlive) continue; // Skip already dead fishes
163
164 double natural_mort_rate = f.naturalMortalityRate(temp);
165 double fishing_mort_rate = fl.fishingMortality(f.length);
166 double mortality_rate = natural_mort_rate + fishing_mort_rate;
167
168 double survival_prob = exp(-mortality_rate*1.0);
169 double catch_prob = fishing_mort_rate/mortality_rate;
170
171 bool f_isAlive = f.isAlive && (runif() <= survival_prob); // set the fish to die probabilistically, if not dead already.
172
173 if (!f_isAlive){
174 if (f.length >= par.lmin){
175 expected_catch += catch_prob * pop.superfish_size * f.weight; // fraction catch_prob of this superfish goes to yield
176 }
177 }
178
179 // ++count;
180 // std::cout << "Sr. / age / length / mu / F / f_isAlive / catch_prob / expt_catch: " << count << " / " << f.age << " / " << f.length << " / " << natural_mort_rate << " / " << fishing_mort_rate << " / " << f_isAlive << " / " << catch_prob << " / " << expected_catch << '\n';
181 }
182
183 if (pop.fishes.size() > 500 && Fc > 0 && expected_catch < 1e-6){
184 std::cout << "Quota is 0 - probably spurious" << std::endl;
185 std::cout << " - nFish = " << pop.fishes.size() << std::endl;
186 std::cout << " - chi = " << fl.chi << std::endl;
187 }
188
189 return expected_catch;
190}

239void Fishery::summarize_population_metrics(){
240 // Calc by-age metrics
241 stock_summary.n_a = pop.aggregateByAge([this](const Fish &f){
242 return (f.isAlive)? pop.superfish_size : 0;
243 });
244
245 stock_summary.w_a = pop.aggregateByAge([this](const Fish &f){
246 return (f.isAlive)? pop.superfish_size*f.weight : 0;
247 });
248 for (int i=0; i<stock_summary.w_a.size(); ++i) stock_summary.w_a[i] /= (stock_summary.n_a[i]+1e-20);
249
250 stock_summary.mat_a = pop.aggregateByAge([this](const Fish &f){
251 return (f.isAlive && f.isMature)? pop.superfish_size : 0;
252 });
253 for (int i=0; i<stock_summary.mat_a.size(); ++i) stock_summary.mat_a[i] /= (stock_summary.n_a[i]+1e-20);
254
255 stock_summary.nfish_ra = stock_summary.n_a[pop.par.recruitmentAge];
256}

259void Fishery::summarize_catch_metrics(bool use_average_weight){
260 // Calc by-age metrics in catch
261 stock_summary.nc_a = pop.aggregateByAge([this](const Fish &f){
262 return (!f.isAlive)? f.fraction_caught*pop.superfish_size : 0;
263 });
264
265 // This calc comes after growth so current weight is inclusive of dw, hence subtract dw/2 to get average weight
266 stock_summary.wc_a = pop.aggregateByAge([use_average_weight, this](const Fish &f){
267 double f_weight_avg = (use_average_weight)? (f.weight - f.delta_weight/2) : f.weight;
268 return (!f.isAlive)? f.fraction_caught*pop.superfish_size*f_weight_avg : 0;
269 });
270 for (int i=0; i<stock_summary.wc_a.size(); ++i) stock_summary.wc_a[i] /= (stock_summary.nc_a[i]+1e-20);
271}

274void Fishery::summarize_spawner_fishery_metrics(const std::vector<double>& spf_summary_before, const std::vector<double>& spf_summary_after, double quota_spf){
275 stock_summary.ssb0 = spf_summary_before[0]; // ssb_before; ///< SSB just before spawning and SPF
276 stock_summary.ssb_spawning = spf_summary_before[1]; // ssb_remaining ///< SSB at spawning time
277 stock_summary.ssb_after_spawning = spf_summary_after[0]; // ssb_before ///< SSB after spawning
278 stock_summary.ssbn = spf_summary_after[1]; // ssb_remaining ///< Final SSB after all mortality

289std::vector<double> Fishery::spawner_fishery(double quota){
290 double ssb_before = pop.calcSSB(pop.par.recruitmentAge);
291 double yield_spf = 0;
292
293 double h_spf = std::clamp(quota/(ssb_before+1e-12), 0.0, 1.0);
294 double p_survival = 1 - h_spf;
295 double ssb_remaining = 0;
296 for (auto &f : pop.fishes) {
297 if (f.isAlive && f.isMature){ // only alive and mature fish are exposed to SPF
298 f.isAlive = f.isAlive && (runif() <= p_survival);
299
300 if (f.isAlive) ssb_remaining += pop.superfish_size*f.weight; // surviving individuals contribute to SSB
301 if (!f.isAlive) yield_spf += pop.superfish_size*f.weight; // dying individuals contribute to yield
302 if (!f.isAlive) f.fraction_caught = 1; // mark fish as caught
303 }
304 }
305
306 return {ssb_before, ssb_remaining, yield_spf, h_spf};
307}

315std::vector<double> Fishery::update(double temp, double K){
316 stock_summary = StockSummary(); // reset stock summary for each year. FIXME: Maybe better to do outside
317
318 // 1. Stock assessment (census) happens here at the beginning of the year, based on which quotas are decided
319 double ssb = pop.calcSSB(pop.par.recruitmentAge);
320 double tsb = pop.calcTSB(pop.par.recruitmentAge);
321 double maturity = pop.calcMaturity(pop.par.recruitmentAge);
322
323 // 2. Calculate total quota (feeding + spawning grounds fishery)
324 double quota = calc_quota(temp);
325 double quota_fgf = quota * (1 - par.rho); // quota for the feeding grounds fishery
326 double quota_spf = quota * par.rho; // quota for the spawning grounds fishery
327
328 // 3. Reproduction and spawning grounds fishery
329 std::vector<double> spf_summary_before = spawner_fishery(quota_spf * par.f_spf_before);
330 std::vector<Fish> recruits = pop.spawn(ssb, tsb, temp, stock_summary);
331 std::vector<double> spf_summary_after = spawner_fishery(quota_spf * (1-par.f_spf_before));
332 summarize_spawner_fishery_metrics(spf_summary_before, spf_summary_after, quota_spf);
333 double yield_spf = spf_summary_before[2] + spf_summary_after[2];
334 double h_spf_eff = spf_summary_before[3] + spf_summary_after[3];
335
336 // 4. Maturation
337 for (auto& f: pop.fishes) f.updateMaturity(temp);
338 summarize_population_metrics(); // l~a, w~a, mat~a, n~a
339
340 // 5. Growth
341 for (auto& f: pop.fishes) f.grow(tsb/1e6, temp); // convert tsb to kT
342
343 // 6. Feeding grounds fishery and natural mortality - should always come after growth to access weights before and after growth
344 std::vector<double> fleet_harvests;
345 bool _use_average_weight = true; // Use average weight (pre and post growth) for yield calcs and catch summary.
346 if (!fleets.empty()) {
347 // Calculate initial chi for all fleets
348 for (auto& fl: fleets) fl.init_chi(pop, -log(1-fl.par.quota*harvest_prop), temp); // initialize chi for the feeding grounds fishery
349
350 // harvest stock (by all fleets)
351 fleet_harvests = pop.get_fished(fleets, {quota_fgf}, temp, _use_average_weight, false);
352 }
353 summarize_catch_metrics(_use_average_weight); // nc~a, wc~a.
354
355 // 7. remove dead fish from population
356 pop.fishes.erase(std::remove_if(pop.fishes.begin(), pop.fishes.end(), [](Fish &f){return !f.isAlive;}), pop.fishes.end());
357
358 // 8. Increment age and advance to new year, and reset delta_weight
359 for (auto& f: pop.fishes) f.set_age(f.age+1);
360
361 // 9. Finally, add recruits to population
362 pop.fishes.insert(pop.fishes.end(), recruits.begin(), recruits.end());
363
364 // 10a. Calculate socioeconomic metrics of feeding grounds fleets
365 std::vector<FleetUtils> fleet_utils_fgf;
366 for (int k=0; k<fleets.size(); ++k){
367 fleet_utils_fgf.push_back(fleets[k].calc_socioeconomics(fleet_harvests[k], K, -9e99, false));
368 }
369
370 // 10b. Calculate socioeconomic metrics of spawner fleets
371 if (spawner_fleets.size() == 0) throw std::runtime_error("No spawner fishery has been added");
372 std::vector<FleetUtils> fleet_utils_spf;
373 for (int k=0; k<spawner_fleets.size(); ++k){
374 double fleet_k_yield = spawner_fleets[k].par.quota * yield_spf;
375 double fleet_k_hspf = spawner_fleets[k].par.quota * h_spf_eff;
376 fleet_utils_spf.push_back(spawner_fleets[k].calc_socioeconomics(fleet_k_yield, -9e99, fleet_k_hspf, true));
377 }
378
379 // 11. Calculate total utils
380 FleetUtils spf_total_utils = std::accumulate(fleet_utils_spf.begin(), fleet_utils_spf.end(), FleetUtils(), std::plus<FleetUtils>());
381 FleetUtils fgf_total_utils = std::accumulate(fleet_utils_fgf.begin(), fleet_utils_fgf.end(), FleetUtils(), std::plus<FleetUtils>());
382 FleetUtils total_utils = spf_total_utils+fgf_total_utils;
383
384 std::vector<double> out = {
385 ssb,
386 total_utils.yield,
387 total_utils.employment_sea + total_utils.employment_shore,
388 total_utils.profit_sea + total_utils.profit_shore,
389 total_utils.effort,
390
391 tsb,
392 maturity,
393 quota,
394 stock_summary.nfish_ra,
395
396 quota_fgf,
397 fgf_total_utils.yield,
398 fgf_total_utils.effort,
399 fgf_total_utils.employment_sea,
400 fgf_total_utils.employment_shore,
401 fgf_total_utils.profit_sea,
402 fgf_total_utils.profit_shore,
403
404 quota_spf,
405 spf_total_utils.yield,
406 spf_total_utils.effort,
407 spf_total_utils.employment_sea,
408 spf_total_utils.employment_shore,
409 spf_total_utils.profit_sea,
410 spf_total_utils.profit_shore,
411
412 stock_summary.ssb0,
413 stock_summary.ssb_spawning,
414 stock_summary.ssb_spawning_ref,
415 stock_summary.ssb_after_spawning,
416 stock_summary.ssb_after_spawning_ref,
417 stock_summary.ssbn,
418 stock_summary.ssbn_ref
419 };
420
421 for (int k=0; k<fleets.size(); ++k){
422 out.push_back(fleet_utils_fgf[k].yield);
423 out.push_back(fleet_utils_fgf[k].effort);
424 out.push_back(fleet_utils_fgf[k].employment_sea);
425 out.push_back(fleet_utils_fgf[k].employment_shore);
426 out.push_back(fleet_utils_fgf[k].profit_sea);
427 out.push_back(fleet_utils_fgf[k].profit_shore);
428 }
429
430 return out;
431}
