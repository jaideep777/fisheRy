130double Stock::avgOverAges(const std::vector<double> &Qa, int amin, int amax, double missing_value){
131 double Q_sum = 0;
132 int nQ = 0;
133 for (int a = amin; a <= amax; ++a){
134 if (Qa[a] != missing_value){
135 Q_sum += Qa[a];
136 nQ += 1;
137 }
138 }
139 if (nQ == 0) return 0;
140 else return Q_sum/nQ;
141}

144inline double runif(double rmin=0, double rmax=1){

149inline double rnorm(double mu=0, double sd=1){
150 double u = runif(), v = runif(); // uniform rn's [0,1] for box-muller
151 double x = sqrt(-2.0*log(u)) * cos(2*M_PI*v);
152 return mu + sd*x;
153}

156vector<Fish> Stock::spawn(double ssb_now, double tsb_now, double temp, StockSummary &stock_summary) {
157 // 3.b.1: Get the number of recruits for each fish
158 // -------------------------------------------------
159 std::vector<double> nrecruits_vec(fishes.size(), 0.0);
160
161 double nrecruits_total = 0;
162 for (int k=0; k<fishes.size(); ++k) {
163 auto &f = fishes[k];
164 if (f.isAlive && f.isMature){ // fish survives the spawning-grounds fishery until actual spawning time
165 // count this fish as spawner and calculate its potential recruits without density effect (for analysis only)
166 stock_summary.nspawners += superfish_size;
167 stock_summary.nrecruits_potential += f.produceRecruits( 0, temp) * superfish_size;
168
169 // Recruit production
170 double nrecruits_fish = f.produceRecruits(ssb_now, temp) * superfish_size;
171 nrecruits_vec[k] = nrecruits_fish;
172 nrecruits_total += nrecruits_fish;
173
174 // Mortality due to spawning
175 double p_survival_spawning = exp(-f.par.Mspawning);
176 f.isAlive = f.isAlive && (runif() <= p_survival_spawning);
177 }
178 }
179
180 //nrecruits *= exp(rnorm(-par.sigmaf*par.sigmaf/2, par.sigmaf));
181 stock_summary.nrecruits_real = std::clamp(nrecruits_total, 1.0, par.rmax);
182 // for (auto& nn : nrecruits_vec) nn = nn*nrecruits_real/(nrecruits_total+1e-20);
183
184 // ** for analysis
185 stock_summary.r0_avg = (ssb_now > 0)? (stock_summary.nrecruits_real * (1 + ssb_now/proto_fish.par.Bhalf) / ssb_now) : -999;
186 stock_summary.factor_dr = stock_summary.nrecruits_real / (stock_summary.nrecruits_potential+1e-12);
187 stock_summary.nrecruits_per_fish = stock_summary.nrecruits_real / stock_summary.nspawners;
188
189 // 3.b.2: Generate recruits, inheriting traits from parents
190 // -------------------------------------------------
191 int n_super_recruits = std::ceil(stock_summary.nrecruits_real / superfish_size);
192
193 vector<Fish> recruits;
194 recruits.reserve(n_super_recruits);
195
196 ++proto_fish.t_birth;
197 std::discrete_distribution<size_t> fitness_dist(nrecruits_vec.begin(), nrecruits_vec.end());
198
199 for (int i=0; i<n_super_recruits; ++i){
200 vector<double> mother_traits = fishes[fitness_dist(generator)].get_traits();
201 vector<double> father_traits = fishes[fitness_dist(generator)].get_traits();
202 vector<double> offspring_traits(mother_traits.size());
203 for (int k=0; k<mother_traits.size(); ++k){
204 offspring_traits[k] = (mother_traits[k] + father_traits[k])/2 + sqrt(proto_fish.trait_variances[k])*proto_fish.trait_scalars[k]*normal_dist(generator);
205 }
206 proto_fish.set_traits(offspring_traits);
207 proto_fish.init(tsb_now/1e6, temp);
208 recruits.push_back(proto_fish);
209 }
210

214
215 return recruits;
216}

220vector<double> Stock::equilibriate_without_fishing(double temp){
221 // start with 1000 age-1 superfish created under the specified temperature
222 init(1000, 0, temp);
223 StockSummary stock_summary;
224
225 std::vector<double> state_t;
226 // run 200 years of population dynamics
227 int nsteps = 200;
228 for (int t=0; t<nsteps; ++t){
229 stock_summary = StockSummary(); // reset stock summary for each year
230
231 double ssb = calcSSB(par.recruitmentAge);
232 double tsb = calcTSB(par.recruitmentAge);
233 double maturity = calcMaturity(par.recruitmentAge);
234
235 // 1. Maturation
236 for (auto& f: fishes) f.updateMaturity(temp);
237
238 // 2. Growth
239 for (auto& f: fishes) f.grow(tsb/1e6, temp); // convert tsb to kT
240
241 // 3. Reproduction
242 vector<Fish> recruits = spawn(ssb, tsb, temp, stock_summary);
243
244 // 4. Mortality
245 for (auto& f: fishes) {
246 double mortality_rate = f.naturalMortalityRate(temp);
247 double survival_prob = exp(-mortality_rate*1.0);
248
249 f.isAlive = f.isAlive && (runif() <= survival_prob); // set the fish to die probabilistically, if not dead already.
250 }
251
252 // 6. remove dead fish from population
253 fishes.erase(std::remove_if(fishes.begin(), fishes.end(), [](Fish &f){return !f.isAlive;}), fishes.end());
254
255 // 7. Increment age and advance to new year
256 for (auto& f: fishes) f.set_age(f.age+1);
257
258 // 8. Finally, add recruits to population
259 fishes.insert(fishes.end(), recruits.begin(), recruits.end());
260
261 // 9. Calculate metrics for analysis
262 state_t.insert(state_t.end(), {
263 ssb,
264 tsb,
265 maturity,
266 stock_summary.nrecruits_real,
267 stock_summary.factor_dr,
268 static_cast<double>(fishes.size())
269 });
270 }
271
272 return state_t;
273}
