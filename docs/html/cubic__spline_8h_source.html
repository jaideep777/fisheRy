15// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
16// Thomas Algorithm to solve Tridiagonal system in O(N) time
17//
18// n is the number of unknowns = matrix size
19// The system solved is as follows:
20//
21// b0 c0 0 0 0 0 0 x0 d0
22// a1 b1 c1 0 0 0 0 x1 d1
23// 0 a2 b2 c2 0 0 0 x2 d2
24// 0 0 a3 b3 c3 0 0 * x3 = d3
25// 0 0 0 a4 b4 c4 0 x4 d4
26// 0 0 0 0 a5 b5 c5 x5 d5
27// 0 0 0 0 0 a6 b6 x6 d6
28//
29// When storing the diagonal (b), upper (c), and lower (a) arrays,
30// array index corresponds to row number in the matrix.
31// Thus, a[0] and c[n-1] are never used.
32//
33// Written by Keivan Moradi, 2014
34// See:
35// https://en.wikibooks.org/wiki/Algorithm_Implementation/Linear_Algebra/Tridiagonal_matrix_algorithm
36// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
37inline void thomas_solve(double* a, double* b, double* c, double* d, int n) {
38 n--; // since we start from x0 (not x1) JAI: Therefore last valid index is n for a,b, n-1 for c
39 c[0] /= b[0];
40 d[0] /= b[0];
41
42 for (int i = 1; i < n; i++) {
43 c[i] /= b[i] - a[i]*c[i-1];
44 d[i] = (d[i] - a[i]*d[i-1]) / (b[i] - a[i]*c[i-1]);
45 }
46
47 d[n] = (d[n] - a[n]*d[n-1]) / (b[n] - a[n]*c[n-1]);
48
49 for (int i = n; i-- > 0;) {
50 d[i] -= c[i]*d[i+1]; // JAI: note, i is in [0, n-1]
51 }
52}

106// This spline interpolation follows the method from
107// https://kluge.in-chemnitz.de/opensource/spline/
108
109// In interval [x{i} - x{i+1}], the function is
110// fi(x) = ai*(x-xi)^3 + bi*(x-xi)^2 + ci*(x-xi) + yi
111// Applying constraints gives:
112// 1. ai = (b{i+1} - b{i})/3/hi hi = x{i+1}-xi
113// 2. ci = (y{i+1}-y{i})/hi - 1/3*(2bi + b{i+1})hi
114// 3. Tridiagonal system for b: for i = [1,n-2]
115// h{i-1}b{i-1}/3 + 2(h{i-1}+h{i})bi/3 + hib{i+1}/3 = (y{i+1}-y{i})/hi - (y{i}-y{i-1})/h{i-1}
116// 4. Boundary conditions: (NEED TO BETTER UNDERSTAND)
117//

177 inline void solve_coeffs_cubic(){
178 int n = npoints;
179 // setting up the matrix and right hand side of the equation system
180 // for the parameters b[]
181 //band_matrix A(n,1,1);
182 std::vector <Float> l(n,0), m(n,0), u(n,0); // lower, middle, and upper diagonals of thr matrix. These hold coefficients of b{i-1}, b{i}, b{i+1} respectively
183 std::vector<double> rhs(n);
184 for(int i=1; i<n-1; i++) {
185 l[i] =1.0/3.0*(m_x[i]-m_x[i-1]);
186 m[i] =2.0/3.0*(m_x[i+1]-m_x[i-1]);
187 u[i] =1.0/3.0*(m_x[i+1]-m_x[i]);
188 rhs[i]=(m_y[i+1]-m_y[i])/(m_x[i+1]-m_x[i]) - (m_y[i]-m_y[i-1])/(m_x[i]-m_x[i-1]);
189 }
190 // boundary conditions, zero curvature b[0]=b[n-1]=0
191 m[0] = 2.0;
192 u[0] = 0.0;
193 rhs[0] = 0.0;
194 m[n-1] = 2.0;
195 l[n-1] = 0.0;
196 rhs[n-1] = 0.0;
197
198 // solve the equation system to obtain the parameters b[]
199 thomas_solve(l.data(),m.data(),u.data(),rhs.data(),n);
200 m_b = rhs;
201
202 // calculate parameters a[] and c[] based on b[]
203 for(int i=0; i<n-1; i++) {
204 m_a[i]=1.0/3.0*(m_b[i+1]-m_b[i])/(m_x[i+1]-m_x[i]);
205 m_c[i]=(m_y[i+1]-m_y[i])/(m_x[i+1]-m_x[i])
206 - 1.0/3.0*(2.0*m_b[i]+m_b[i+1])*(m_x[i+1]-m_x[i]);
207 }
208 // for the right boundary we define
209 // f_{n-1}(x) = b*(x-x_{n-1})^2 + c*(x-x_{n-1}) + y_{n-1}
210 double h=m_x[n-1]-m_x[n-2];
211 // m_b[n-1] is determined by the boundary condition (turns out to be 0; = 0 for 0 curvature at xn-1)
212 m_a[n-1] = 0.0;
213 m_c[n-1]=3.0*m_a[n-2]*h*h+2.0*m_b[n-2]*h+m_c[n-2]; // = f'_{n-2}(x_{n-1})
214 }

217 inline void solve_coeffs_constrained_cubic(){
218 int n = npoints;
219
220 std::vector <double> m(n);
221 for (int i=1; i<n-1; ++i){
222 double fplus = (m_y[i+1]-m_y[i])/(m_x[i+1]-m_x[i]);
223 double fminus = (m_y[i]-m_y[i-1])/(m_x[i]-m_x[i-1]);
224 if (fplus*fminus < 0) m[i] = 0;
225 else m[i] = 2/(1/fplus + 1/fminus); // harmonic mean will tend towards smaller derivative
226 //else m[i] = (fplus+fminus)/2;
227 }
228 // Boundary conditions: zero curvature at x0 and xN-1
229 m[0] = 3*(m_y[1]-m_y[0])/(m_x[1]-m_x[0])/2 - m[1]/2;
230 m[n-1] = 3*(m_y[n-1]-m_y[n-2])/(m_x[n-1]-m_x[n-2])/2 - m[n-2]/2;
231
232 for (int i=0; i<n-1; ++i){
233 m_c[i] = m[i];
234 double hi = m_x[i+1]-m_x[i];
235 m_b[i] = ( 3*(m_y[i+1]-m_y[i] - m[i]*hi) - hi*(m[i+1]-m[i]))/hi/hi;
236 m_a[i] = (-2*(m_y[i+1]-m_y[i] - m[i]*hi) + hi*(m[i+1]-m[i]))/hi/hi/hi;
237 }
238 m_b[n-1] = 0;
239 m_c[n-1] = m[n-1];
240 }
