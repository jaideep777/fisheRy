125double Fish::naturalMortalityRate(double temp) const{
126 double rate;
127 if (age > par.amax) return 1e20; // FIXME: use inf
128 else {
129 if (par.mortality_model == MortalityModel::Dankel22){
130 if (isMature) rate = par.mam[age];
131 else rate = par.mai[age];
132 return rate;
133 }
134 else if (par.mortality_model == MortalityModel::Bioenergetic){
135 //return fish::natural_mortality(length, temp, par.M0, par.gamma3, par.alpha3, par.Lref, par.Tref, par.cT);
136 // Note: Mortality due to spawning is not included here, it is treated separately in the reproduction part of the population update.
137 return (par.M0 +
138 par.alpha3 * pow(length / par.Lref, par.gamma3) +
139 par.alpha4*(par.alpha1*par.alpha1 - par.alpha1_ref*par.alpha1_ref) +
140 par.alpha5*(par.gsi - par.gsi_ref)
141 ) * pow(temp/par.Tref, par.cT);
142 }
143 else if (par.mortality_model == MortalityModel::Empirical){
144 if (mort_fn_spline.npoints == 0) throw std::runtime_error("Natural mortility function is set to empirical but spline is not set.");
145 return natural_mort_scalar * (mort_fn_spline.eval(length) - mort_fn_spline.eval(1000)) + par.M0;
146 }
147 else{
148 throw std::runtime_error("Invalid mortality model specified");
149 }
150 }

189void Fish::grow(double tsb, double temp){
190 double weight_before = weight;
191 if (par.growth_model == GrowthModel::Dankel22){
192 // do nothing. age is incremented by population update
193 gsi_effective = par.gsi; // required if new fecundity model is used in combination with old growth model
194 }
195 else if (par.growth_model == GrowthModel::Bioenergetic){
196 double tsb_ano = tsb - par.tsbmean;
197 double temp_ano = temp - par.Tmean;
198
199 // This is generalized increment l2^y1y2 - l1^y1y2
200 double dl_nonlinear = fish::dl_power(tsb_ano, temp_ano, par.gamma1, par.gamma2, par.alpha1, par.alpha2, par.beta1, par.beta2);
201 double dl_pot_nonlinear = fish::dl_power(-par.tsbmean, temp_ano, par.gamma1, par.gamma2, par.alpha1, par.alpha2, par.beta1, par.beta2);
202
203 // Calculate new length if there was no reproductive investment
204 double lnew, lnew_pot; // Real and potential increments are with and without density effects
205 lnew = fish::length_juvenile(length, dl_nonlinear, par.gamma1, par.gamma2);
206 lnew_pot = fish::length_juvenile(length, dl_pot_nonlinear, par.gamma1, par.gamma2);
207
208 // Calculate linear increment (real and potential)
209 dl_real = lnew - length;
210 dl_potential = lnew_pot - length; // This is only for assessing density effect
211 //cout << "tsb_ano = " << tsb << " / " << par.tsbmean << ", fac = " << dl_real << " / " << dl_potential << endl;
212
213 // add environmental noise on real growth
214 dl_real_stochastic = dl_real*std::clamp(rnorm(1, par.growth_noise_sd), 0.0, 5.0);
215
216 // Recalculate new length based on linear length increment
217 double lnew_stochastic = length + dl_real_stochastic;
218 double dl_juvenile_nonlinear_stochastic = pow(lnew_stochastic, par.gamma1*par.gamma2) - pow(length, par.gamma1*par.gamma2);
219
220 // If fish is mature, length increment is reduced due to reproductive investment
221 if (isMature){
222 lnew_stochastic /= pow(1 + par.gamma1*par.gsi, 1/(par.gamma1*par.gamma2));
223 lnew_stochastic = std::max(lnew_stochastic, length);
224 }
225
226 // Effective gsi - should be 0 for juveniles and par.gsi for adults
227 gsi_effective = fish::gsi(lnew_stochastic, length, dl_juvenile_nonlinear_stochastic, par.gamma1, par.gamma2, par.alpha1, par.alpha2);
228
229 // Set new length
230 set_length(lnew_stochastic);
231 }
232 else{
233 throw std::runtime_error("Invalid growth model specified");
234 }
235 delta_weight = weight - weight_before;
238}

277double Fish::produceRecruits(double ssb, double temp){
278 double temp_ano = temp - par.Tref;
279 if (par.recruitment_model == RecruitmentModel::BevertonHoltDirect){
280 double recruits = par.r0 * weight * 1 / (1 + ssb/par.Bhalf);
281 return recruits;
282 }
283 else if (par.recruitment_model == RecruitmentModel::RickerDirect){
284 double recruits = par.r0 * weight * exp(par.beta4*temp_ano) * pow(2, -ssb/par.Bhalf);
285 return recruits;
286 }
287 else if (par.recruitment_model == RecruitmentModel::BevertonHoltBioenergetic){
288 double eggs = fish::fecundity(weight, par.delta, gsi_effective); // Total eggs produced
289 double recruits = eggs * par.s0 * 1 / (1 + ssb/par.Bhalf); // offspring surviving density-dependent recruitment
290 return recruits;
291 }
292 else if (par.recruitment_model == RecruitmentModel::RickerBioenergetic){
293 double eggs = fish::fecundity(weight, par.delta, gsi_effective); // Total eggs produced
294 double recruits = eggs * par.s0 * exp(par.beta4*temp_ano) * pow(2, -ssb/par.Bhalf); // offspring surviving density-dependent recruitment
295 return recruits;
296 }
297 else{
298 throw std::runtime_error("Invalid recruitment model specified");
299 }
