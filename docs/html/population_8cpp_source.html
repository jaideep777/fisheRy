254vector<double> Population::noFishingEquilibriate(double temp){

256 auto par_back = par;
257
258 // disable fishing and env stochasticity
259 set_harvestProp(0);
260 par.sigmaf = 0; // no env stochasticity while calculating carrying capacity
261 par.simulate_bio_only = true;
262
263 // start with 1000 age-1 superfish created under the specified temperature
264 init(1000, temp);
265
266 // run 200 years of population dynamics
267 int nsteps = 200;
268 std::vector<double> state_t;
269 for (int t=0; t<nsteps; ++t){

272

782
783 double ssb_after_spawning_ref = ssb0*exp(-proto_fish.par.Mspawning)*(1-par.f_spf_before*h_spf);
784 if (verbose) cout << "ssb after spawning = " << ssb_after_spawning << " / " << ssb_after_spawning_ref << endl;
785
786 // 3c. post-spawning part of the SPF
787 double p_survival_spf_after = (1-h_spf)/(1 - par.f_spf_before*h_spf);
788 for (int k=0; k<fishes.size(); ++k) {
789 auto &f = fishes[k];
790 if (f.isAlive && f.isMature){ // only mature fish are exposed to SPF
791 f.isAlive = f.isAlive && (runif() <= p_survival_spf_after);
792
793 if (!f.isAlive) yield_spf += par.n * f.weight;
794 if (!f.isAlive) f.isCaught = true; // mark fish as caught
795
796 }
797 }
798
799 double ssbn = calcSSB(par.recruitmentAge);
800
801 double yield_spf_ref = ssb0*h_spf*(par.f_spf_before + (1-par.f_spf_before)*exp(-proto_fish.par.Mspawning));
802 double ssbn_ref = ssb0*(1-h_spf)*exp(-proto_fish.par.Mspawning);
803
804 // 4. Mortality
805
806 // 4.1. Effort Dynamics
807 double F_req = par.Fc; // total control fishing mortality rate should be used in effort dynamics
808 double F_real = F_req;
809 double E_req = 0, E_real = 0;
810 double D_sea_req = 0, D_sea_real = 0;
811 double Nrel = 0;
812
813 if (!par.simulate_bio_only){
814 if (par.h > 0){
815 Nrel = (K_fishableBiomass > 0)? fishableBiomass() / K_fishableBiomass : 1e-20;
816
817 // Effort covers spawning and feeding grounds, so
818 // we use the total control fishing mortality rate and the total natural mortality rate
819 E_req = (Nrel < 1e-20)? 0 : effort1(Nrel, F_real, Mort_ref);
820 //pow(Nrel, 1-par.b) * F * (exp(-(F+M)*(1-par.b))-1) / (par.q*(F+M)*(par.b-1));
821 D_sea_req = par.dsea * E_req;
822 D_sea_real = D_sea_req / (1 + D_sea_req/par.dmax);
823
824 E_real = D_sea_real / par.dsea;
825 // Solve for F_real - ignore for now
826 // F_real = pn::zero(0, F_req, [E_real, Nrel, temp, this](double F){ return (E_real - effort(Nrel, F, temp));}, 1e-6).root;
827 F_real = F_req;
828 }
829 }
830
831 // implement natural+fishing mortality in feeding grounds over the rest of the year, and calculate yield
832 double tsb_before_mort = calcTSB(par.recruitmentAge);
833 double yield = 0, to_sea_bed = 0;
834 double survival_mean = 0, n_survival_mean = 0;
835 for (auto& f : fishes){
836 if (f.isAlive){
837 double fishing_mort_rate = chi*fishingMortalityRef(f.length); //*(F_real/(F_req+1e-20)); // the factor F_real/F_req is needed if effort limitation is used
838 double natural_mort_rate = f.naturalMortalityRate(temp); // This does not (should not) include spawning-related mortality
839 double mortality_rate = natural_mort_rate + fishing_mort_rate; // post-spawning mortality rate is same for mature and immature individuals
840 double survival_prob = exp(-mortality_rate*1.0); // mortality in feeding grounds (post-spawning), over full year.
841 survival_mean += survival_prob;
842 n_survival_mean += 1;
843
844 f.isAlive = f.isAlive && ((rand() / double(RAND_MAX)) <= survival_prob); // set the fish to die probabilistically, if not dead already.
845
846 if (!f.isAlive){
847 f.isCaught = runif() < fishing_mort_rate/mortality_rate; // check if fish is caught or goes to sea bed!
848
849 if (f.isCaught) yield += par.n*f.weight; // if caught, add to yield
850 else to_sea_bed += par.n*f.weight; // else, goes to sea bed
851 }
852 }
853 }
854 survival_mean /= n_survival_mean;
855
856 double tsb_after_mort = calcTSB(par.recruitmentAge);
857
858 // Calc by-age metrics in catch
859 pop_summary.nc_a = aggregateByAge([this](const Fish &f){
860 return (!f.isAlive && f.isCaught)? par.n : 0;
861 });
862
863 pop_summary.wc_a = aggregateByAge([this](const Fish &f){
864 return (!f.isAlive && f.isCaught)? par.n*f.weight : 0;
865 });
866 for (int i=0; i<pop_summary.wc_a.size(); ++i) pop_summary.wc_a[i] /= (pop_summary.nc_a[i]+1e-20);
867
868
869 // remove dead fish from population
870 fishes.erase(std::remove_if(fishes.begin(), fishes.end(), [](Fish &f){return !f.isAlive;}), fishes.end());
871 int nfish_after_mort = fishes.size();
872
873 // 5. Increment age and advance to new year
874 for (auto& f: fishes){
875 f.set_age(f.age+1);
876 }
877
878 // 6. Finally, add recruits to population
879 fishes.insert(fishes.end(), recruits.begin(), recruits.end());
880
881 // calculate employment
882 double emp_sea = D_sea_req;
883 double emp_shore = par.dshr * yield;
884
885 // calculate profit for the year
886 double profit_sea = 0, profit_shr = 0;
887 if (!par.simulate_bio_only){
888 //if (par.h > 0){
889 profit_sea = yield*par.price_sea*(1-par.fee_ratio) - par.scale_catch*(D_sea_req*par.salary_sea + E_req*par.variable_costs_sea + par.fixed_costs_sea);
890 profit_shr = yield*(par.price_shore - par.price_sea) - yield*par.dshr * par.salary_shore - par.scale_catch*par.fixed_costs_shore;
891 //}
892 }
893
894 if (verbose) cout << "year = " << current_year
895 << " | TSB(MT) = " << tsb/1e9 << ", SSB(MT) = " << ssb/1.0e9
896 << ", recruits = " << nrecruits_real << "/" << std::accumulate(nrecruits_vec.begin(), nrecruits_vec.end(), 0.0)
897 << ", maturity = " << maturity
898 << ", survival = " << nfish_after_mort << "/" << nfish_start
899 << ", survival_prob = " << survival_mean << " --> " << double(nfish_after_mort)/nfish_start
900 << ", N_rel = " << Nrel
901 << ", F_real = " << F_real << "(" << F_real/(F_req+1e-20)*100
902 << "%), r0_avg = " << r0_avg
903 << ", % harvest = " << yield/tsb
904 << ", dg/dr = " << factor_dg << "/" << factor_dr
905 << ", yield = " << yield
906 << "\n";
907
908 ++current_year;
909 return {ssb, yield, emp_sea+emp_shore, profit_sea+profit_shr, emp_sea, emp_shore, profit_sea, profit_shr, tsb, r0_avg, nrecruits_real, nfish_ra, static_cast<double>(nfish()), factor_dg, factor_dr, lmax, length90, survival_mean, maturity, Nrel,
910 ssb_spawning, ssb_spawning_ref, ssb_after_spawning, ssb_after_spawning_ref, ssbn, ssbn_ref, yield_spf, yield_spf_ref,
911 tsb_before_mort, tsb_after_mort, to_sea_bed,
912 chi, Fref_ref, Mort_ref, Mat_ref, F_spf
913 };
914}
