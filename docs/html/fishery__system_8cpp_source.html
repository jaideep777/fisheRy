98// This is with the new definition of harvest proportion, i.e., a fraction h of fishable biomass
99double Fishery::calc_quota(double temp){
100 if (fleets.empty()) {
101 throw std::runtime_error("Fishery: No fleets defined, cannot calculate quota.");
102 }
103
104 auto& fl = fleets[0]; // Since lmin is the same across all fleets, we use the first fleet to calculate quota
105 double Fc = -log(1-harvest_prop);
106
107 // Calculate chi of the fleet to implement mortality
108 fl.chi = Fc/fl.FishingMortalityRef_avgl(200, 100);
109
110 // Calculate expected Catch with the given chi
111 // Note: Be careful to not modify the fishes vector here
112 double expected_catch = 0, to_sea_bed = 0;
113 for (const auto& f: pop.fishes) { // const auto& ensures we do not modify the fish
114 if (!f.isAlive) continue; // Skip already dead fishes
115
116 double natural_mort_rate = f.naturalMortalityRate(temp);
117 double fishing_mort_rate = fl.fishingMortality(f.length);
118 double mortality_rate = natural_mort_rate + fishing_mort_rate;
119
120 double survival_prob = exp(-mortality_rate*1.0);
121
122 bool f_isAlive = f.isAlive && (runif() <= survival_prob); // set the fish to die probabilistically, if not dead already.
123
124 if (!f_isAlive){
125 bool f_isCaught = runif() < fishing_mort_rate/mortality_rate; // check if fish is caught or goes to sea bed!
126
127 if (f_isCaught) expected_catch += pop.superfish_size*f.weight; // if caught, add to yield
128 else to_sea_bed += pop.superfish_size*f.weight; // else, goes to sea bed
129 }
130
131 }
132
133 return expected_catch;
134}

388void Fishery::summarize_population_metrics(){
389 // Calc by-age metrics
390 stock_summary.n_a = pop.aggregateByAge([this](const Fish &f){
391 return (f.isAlive)? pop.superfish_size : 0;
392 });
393
394 stock_summary.w_a = pop.aggregateByAge([this](const Fish &f){
395 return (f.isAlive)? pop.superfish_size*f.weight : 0;
396 });
397 for (int i=0; i<stock_summary.w_a.size(); ++i) stock_summary.w_a[i] /= (stock_summary.n_a[i]+1e-20);
398
399 stock_summary.mat_a = pop.aggregateByAge([this](const Fish &f){
400 return (f.isAlive && f.isMature)? pop.superfish_size : 0;
401 });
402 for (int i=0; i<stock_summary.mat_a.size(); ++i) stock_summary.mat_a[i] /= (stock_summary.n_a[i]+1e-20);
403}

419std::vector<double> Fishery::update(double temp){
420 stock_summary = StockSummary(); // reset stock summary for each year. FIXME: Maybe better to do outside
421
422 double ssb = pop.calcSSB(pop.par.recruitmentAge);
423 double tsb = pop.calcTSB(pop.par.recruitmentAge);
424 double maturity = pop.calcMaturity(pop.par.recruitmentAge);
425
426 // 1. Maturation
427 for (auto& f: pop.fishes) f.updateMaturity(temp);
428 summarize_population_metrics();
429
430 // 2. Growth
431 for (auto& f: pop.fishes) f.grow(tsb/1e6, temp); // convert tsb to kT
432
433 // 3. Calculate total quota (feeding + spawning grounds fishery)
434 double quota = calc_quota(temp);
435 double quota_fgf = quota * (1 - par.rho); // quota for the feeding grounds fishery
436 double quota_spf = quota * par.rho; // quota for the spawning grounds fishery
437
438 // 4. Reproduction
439 std::vector<Fish> recruits = pop.spawn(ssb, tsb, temp, stock_summary);
440
441 // 5. Feeding grounds fishery and natural mortality
442 double yield = 0, effort = 0;
443 if (!fleets.empty()) {
444 fleets[0].init_chi(pop, -log(1-harvest_prop), temp); // initialize chi for the feeding grounds fishery
445 std::vector<double> harvest_out = fleets[0].harvest(pop, quota_fgf, temp, false);
446 yield = harvest_out[0]; // total yield from the feeding grounds fishery
447 effort = 0; // fleets[0].effort_constantC(fleets[0].par.q, fleets[0].par.b, pop.fishableBiomass());
448 }
449 summarize_catch_metrics();
450
451 // 6. remove dead fish from population
452 pop.fishes.erase(std::remove_if(pop.fishes.begin(), pop.fishes.end(), [](Fish &f){return !f.isAlive;}), pop.fishes.end());
453
454 // 7. Increment age and advance to new year
455 for (auto& f: pop.fishes) f.set_age(f.age+1);
456
457 // 8. Finally, add recruits to population
458 pop.fishes.insert(pop.fishes.end(), recruits.begin(), recruits.end());
459
460 // 9. Calculate metrics for analysis
461 return {
462 ssb,
463 tsb,
464 maturity,
465 quota_fgf,
466 yield,
467 effort
468 };
469}
