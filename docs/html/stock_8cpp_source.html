148vector<Fish> Stock::spawn(double ssb_now, double tsb_now, double temp, StockSummary &stock_summary) {
149 // 3.b.1: Get the number of recruits for each fish
150 // -------------------------------------------------
151 std::vector<double> nrecruits_vec(fishes.size(), 0.0);
152
153 double nrecruits_total = 0;
154 for (int k=0; k<fishes.size(); ++k) {
155 auto &f = fishes[k];
156 if (f.isAlive && f.isMature){ // fish survives the spawning-grounds fishery until actual spawning time
157 // count this fish as spawner and calculate its potential recruits without density effect (for analysis only)
158 stock_summary.nspawners += superfish_size;
159 stock_summary.nrecruits_potential += f.produceRecruits( 0, temp) * superfish_size;
160
161 // Recruit production
162 double nrecruits_fish = f.produceRecruits(ssb_now, temp) * superfish_size;
163 nrecruits_vec[k] = nrecruits_fish;
164 nrecruits_total += nrecruits_fish;
165
166 // Mortality due to spawning
167 double p_survival_spawning = exp(-f.par.Mspawning);
168 f.isAlive = f.isAlive && (runif() <= p_survival_spawning);
169 }
170 }
171
172 double nrecruits_before_noise = nrecruits_total;
173 double noise_multiplier = exp(rnorm(-par.sigmaf*par.sigmaf/2, par.sigmaf));
174 noise_multiplier = clamp(noise_multiplier, 1e-3, 10.0);
175 nrecruits_total *= noise_multiplier;
176 stock_summary.nrecruits_real = std::clamp(nrecruits_total, 1.0, par.rmax);
177 // for (auto& nn : nrecruits_vec) nn = nn*nrecruits_real/(nrecruits_total+1e-20);
178
179 // ** for analysis
180 stock_summary.r0_avg = (ssb_now > 0)? (stock_summary.nrecruits_real * (1 + ssb_now/proto_fish.par.Bhalf) / ssb_now) : -999;
181 stock_summary.factor_dr = stock_summary.nrecruits_real / (stock_summary.nrecruits_potential+1e-12);
182 stock_summary.nrecruits_per_fish = stock_summary.nrecruits_real / stock_summary.nspawners;
183
184 // 3.b.2: Generate recruits, inheriting traits from parents
185 // -------------------------------------------------
186 int n_super_recruits = std::ceil(stock_summary.nrecruits_real / superfish_size);
187
188 vector<Fish> recruits;
189 try{
190 recruits.reserve(n_super_recruits);
191 }
192 catch(std::exception &e){
193 std::cout << "Could not reserve recruits vec - required size: " << n_super_recruits << " / " << e.what() << std::endl;
194 std::cout << "nrecruits_before_noise / nrecruits_total = " << nrecruits_before_noise << " / " << nrecruits_total << std::endl;
195 throw e;
196 }
197
198 ++proto_fish.t_birth;
199 std::discrete_distribution<size_t> fitness_dist(nrecruits_vec.begin(), nrecruits_vec.end());
200
201 for (int i=0; i<n_super_recruits; ++i){
202 vector<double> mother_traits = fishes[fitness_dist(rng)].get_traits();
203 vector<double> father_traits = fishes[fitness_dist(rng)].get_traits();
204 vector<double> offspring_traits(mother_traits.size());
205 for (int k=0; k<mother_traits.size(); ++k){
206 offspring_traits[k] = (mother_traits[k] + father_traits[k])/2 + sqrt(proto_fish.trait_variances[k])*proto_fish.trait_scalars[k]*rnorm();
207 }
208 proto_fish.set_traits(offspring_traits);
209 proto_fish.init(tsb_now/1e6, temp);
210 recruits.push_back(proto_fish);
211 }
212
213 // FIXME: uncomment these
214 // stock_summary.ssb_after_spawning = calcSSB(par.recruitmentAge);
215 // stock_summary.ssb_after_spawning_ref = stock_summary.ssb0*exp(-proto_fish.par.Mspawning)*(1-par.f_spf_before*h_spf);
216
217 return recruits;
218}

222vector<double> Stock::equilibriate_without_fishing(double temp, int nsteps){
223 // start with 1000 age-1 superfish created under the specified temperature
224 init(1000, 0, temp);
225 StockSummary stock_summary;
226
227 std::vector<double> state_t;
228 // run nsteps years of population dynamics
229 for (int t=0; t<nsteps; ++t){
230 stock_summary = StockSummary(); // reset stock summary for each year
231
232 double ssb = calcSSB(par.recruitmentAge);
233 double tsb = calcTSB(par.recruitmentAge);
234 double maturity = calcMaturity(par.recruitmentAge);
235
236 // 1. Maturation
237 for (auto& f: fishes) f.updateMaturity(temp);
238
239 // 2. Growth
240 for (auto& f: fishes) f.grow(tsb/1e6, temp); // convert tsb to kT
241
242 // 3. Reproduction
243 vector<Fish> recruits = spawn(ssb, tsb, temp, stock_summary);
244
245 // 4. Mortality
246 for (auto& f: fishes) {
247 double mortality_rate = f.naturalMortalityRate(temp);
248 double survival_prob = exp(-mortality_rate*1.0);
249 f.isAlive = f.isAlive && (runif() <= survival_prob); // set the fish to die probabilistically, if not dead already.
250 }
251
252 // 6. remove dead fish from population
253 fishes.erase(std::remove_if(fishes.begin(), fishes.end(), [](Fish &f){return !f.isAlive;}), fishes.end());
254
255 // 7. Increment age and advance to new year
256 for (auto& f: fishes) f.set_age(f.age+1);
257
258 // 8. Finally, add recruits to population
259 fishes.insert(fishes.end(), recruits.begin(), recruits.end());
260
261 // 9. Calculate metrics for analysis
262 state_t.insert(state_t.end(), {
263 ssb,
264 tsb,
265 maturity,
266 stock_summary.nrecruits_real,
267 stock_summary.factor_dr,
268 static_cast<double>(fishes.size())
269 });
270 }
271
272 return state_t;
273}

276 double window_dt = 0.1;
277 Fleet& ref_fleet = fleets[0];
278
279 // Calculate total quota
280 double quota = std::accumulate(quotas.begin(), quotas.end(), 0.0);
281
282 // Live fish to sample per window
283 int n_alive = std::accumulate(fishes.begin(), fishes.end(), 0,
284 [this](int sum, const Fish& f) { return sum + ((f.isAlive) ? 1 : 0); }
285 );
286 int window_n = std::ceil(window_dt*n_alive);
287
288 // Randomize fishes vector
289 std::shuffle(fishes.begin(), fishes.end(), rng);
290
291 // Total available biomass to be sampled
292 double B = ref_fleet.biomassFishable(*this, 0, use_average_weight);
293
294 // Clear window props vector in all fleets
295 for (auto& fl : fleets) fl.window_props_vec.clear(); // clear old data in windows
296
297 double B_sampled = 0, yield_expected = 0;
298 double yield = 0, to_sea_bed = 0;
299 int windows_sampled = 0;
300
301 // fleet-wise yields
302 std::vector<double> yields(fleets.size(), 0);
303
304 WindowProps wp_total_debug;
305 std::vector<WindowProps> wps_per_fleet(fleets.size()); // create a new WindowProps object for each fleet
306 std::vector<double> progress;
307
308 int live_fish_count = 0;
309 for (auto& f : fishes){
310 if (!f.isAlive) continue; // skip dead fish
311
312 ++live_fish_count;
313
314 // If this is beginning of window, update start-of-window window_props
315 if (live_fish_count % window_n == 1){
316 for (int k=0; k<fleets.size(); ++k){
317 wps_per_fleet[k].B_start = B - yield - to_sea_bed; // biomass at start of window is total biomass - biomass died so far
318 wps_per_fleet[k].chi = fleets[k].chi;
319 }
320 }
321
322 double f_weight = (use_average_weight)? (f.weight - f.delta_weight/2) : f.weight;
323 double fishability_f = ref_fleet.fishability(f.length);
324
325 // fishable biomass of current fish and fishable biomass sampled until this point
326 double B_sampled_t = fishability_f * f_weight * superfish_size;
327 B_sampled += B_sampled_t;
328 wp_total_debug.B_sampled += B_sampled_t;
329 if (B_sampled > B+1){
330 std::cout << "Sampled fishable biomass exceeds total fishable biomass: " << B_sampled << " / " << B << std::endl;
331 throw std::runtime_error("");
332 }
333
334 // expected yield until this point
335 yield_expected = (B_sampled/B) * quota;
336
337 // fleet-wise fishing mort rates
338 std::vector<double> fishing_morts;
339 fishing_morts.reserve(fleets.size());
340 for (auto& fl : fleets) fishing_morts.emplace_back(fl.fishingMortality(f.length));
341
342 // total fishing mort and natural mort rates
343 double fishing_mort_rate = std::accumulate(fishing_morts.begin(), fishing_morts.end(), 0.0); // total fishing mort rate from all fleets
344 double natural_mort_rate = f.naturalMortalityRate(temp); // This does not (should not) include spawning-related mortality
345 double mortality_rate = natural_mort_rate + fishing_mort_rate; // post-spawning mortality rate is same for mature and immature individuals
346 double survival_prob = exp(-mortality_rate*1.0); // mortality in feeding grounds (post-spawning), over full year. Note that survival prob must be annualized because this fish will be iterated over only once
347
348 if (debug && natural_mort_rate > 100) std::cout << "Unusually high M: " << f.age << " / " << f.length << " / " << natural_mort_rate << '\n';
349
350 // Set fish to die based on total mort
351 f.isAlive = f.isAlive && (runif() <= survival_prob); // set the fish to die probabilistically, if not dead already.
352
353 if (!f.isAlive){
354 f.fraction_caught = fishing_mort_rate/mortality_rate; // what fraction of the superfish goes to yield (vs seabed)?
355
356 double yield_t = f.fraction_caught * superfish_size * f_weight; // catch_frac fraction goes to yield
357 yield += yield_t;
358 wp_total_debug.yield += yield_t;
359 to_sea_bed += (1-f.fraction_caught) * superfish_size * f_weight; // remaining fraction goes to sea bed
360
361 // Allot yield to each fleet
362 for (int k=0; k<fleets.size(); ++k){
363 double catch_frac_fleet = fishing_morts[k]/mortality_rate;
364 double yield_t_k = catch_frac_fleet * superfish_size * f_weight;
365 wps_per_fleet[k].yield += yield_t_k;
366 yields[k] += yield_t_k;
367 }
368 }
369
370 // Fill remaining window_props for each fleet
371 for (int k=0; k<fleets.size(); ++k){
372 wps_per_fleet[k].B_sampled += B_sampled_t;
373 wps_per_fleet[k].chi = fleets[k].chi;
374 if (f.age <= f.par.amax){ // Need to skip age 31 fish as M is 1e20
375 wps_per_fleet[k].F_fishable += fishability_f * fishing_morts[k];
376 wps_per_fleet[k].M_fishable += fishability_f * natural_mort_rate;
377 wps_per_fleet[k].n_fishable += fishability_f * 1;
378 }
379 }
380
381 // When we've sampled all live fish in this window, perform window closing operations
382 if (live_fish_count % window_n == 0){
383 ++windows_sampled;
384
385 double bs_remainder = B - B_sampled;
386
387 // For each fleet
388 for (int k=0; k<fleets.size(); ++k){
389 // Compute average window properties
390 wps_per_fleet[k].F_fishable /= wps_per_fleet[k].n_fishable;
391 wps_per_fleet[k].M_fishable /= wps_per_fleet[k].n_fishable;
392 wps_per_fleet[k].C_rate = wps_per_fleet[k].yield / window_dt;
393
394 // Push window props into vector
395 fleets[k].window_props_vec.push_back(wps_per_fleet[k]);
396
397 // update chi
398 double yield_remainder = quotas[k] - yields[k];
399 fleets[k].update_chi_implicit(yield_remainder, bs_remainder);
400
401 // reset WindowProps
402 wps_per_fleet[k] = WindowProps();
403 }
404
405 }
406
407 if (return_progress){
408 progress.insert(progress.end(),
409 {
410 static_cast<double>(f.age),
411 B,
412 B_sampled,
413 yield,
414 yield_expected
415 });
416
417 for (int k=0; k<fleets.size(); ++k){
418 progress.insert(progress.end(),
419 {
420 fleets[k].chi,
421 wps_per_fleet[k].chi,
422 wps_per_fleet[k].B_sampled,
423 wps_per_fleet[k].B_start,
424 wps_per_fleet[k].yield,
425 wps_per_fleet[k].F_fishable,
426 // Below effort calc is only for debugging
427 fleets[k].effort_constantC(fleets[k].par.q, fleets[k].par.b, B)*fleets[k].par.dsea,
428 fleets[k].effort_constantF(fleets[k].par.q, fleets[k].par.b, B)*fleets[k].par.dsea
429 });
430 }
431 }
432
433 if (std::isnan(yield)){
434 std::cout << "Yield is NaN. Dump = \n";
435 for (int i=0; i<progress.size(); ++i){
436 if (i > 0 & i % (5 + 8*fleets.size()) == 0) std::cout << "\n";
437 cout << progress[i] << " ";
438 }
439 std::cout << std::endl;
440
441 std::cout << "Window props = \n";
442 for (auto& w : fleets[0].window_props_vec){
443 cout << w.B_sampled << " " << w.yield << " " << w.chi << std::endl;
444 }
445 }
446 }
447
448 if (return_progress) return progress;
449 else return yields;
450}
