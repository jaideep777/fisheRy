187std::vector<double> Fleet::cummulativeFishingMortalityRef(const Stock &stock, double min_age){
188 double wF_below_lmin = 0, wF_above_lmin = 0, w_sum = 0;
189 for (auto& f : stock.fishes){
190 if (f.age < min_age || !f.isAlive) continue; // skip fish below min age or dead
191
192 double w = fishability(f.length);
193 w_sum += w;
194 if (f.length < par.lmin) wF_below_lmin += w * fishingMortalityRef(f.length);
195 else wF_above_lmin += w * fishingMortalityRef(f.length);
196 }
197 return {wF_below_lmin, wF_above_lmin, w_sum, (wF_below_lmin+wF_above_lmin)/(w_sum+1e-20)};
198 // ^ sum(wF) ^ sum(wF) ^ sum(w) ^ average F weighted by w
199}

256void Fleet::update_chi(const std::vector<double>& chi_in_windows,
257 const std::vector<double>& yield_in_windows,
258 const std::vector<double>& bs_in_windows,
259 double yield_remainder, double bs_remainder){
260 // 3a. calibrate yield model (y = Bs * f(X))
261 linregresult res;
262 std::vector<double> y(yield_in_windows.size());
263 if (par.control_model == "exp"){
264 // exponential model: y = Bs (1-e^-kX) --> -log(1-y/Bs) = kX
265 std::transform(yield_in_windows.begin(), yield_in_windows.end(),
266 bs_in_windows.begin(), y.begin(),
267 [](double yield, double bs) {
268 if (bs == 0) return 0.0;
269 if (yield >= bs) return 25.0;
270 return -log(1 - (yield / bs));
271 });
272
273 res = linreg0(chi_in_windows, y, debug);
274 }
275 else if (par.control_model == "linear"){
276 // linear model: y = Bs (k X) --> y/Bs = kX
277 std::transform(yield_in_windows.begin(), yield_in_windows.end(),
278 bs_in_windows.begin(), y.begin(),
279 [](double yield, double bs) {
280 return (bs == 0)? 0 : yield / bs;
281 });
282
283 res = linreg0(chi_in_windows, y, debug);
284 }
285 else {
286 throw std::runtime_error("Unsopported control model");
287 }
288
289 // 3b. predict new chi
290 if (par.control_model == "exp"){
291 // exponential model: y = Bs (1-e^-kX)
292 // std::cout << "using exp model" << std::endl;
293 if (yield_remainder >= bs_remainder) chi = par.max_chi;
294 else if (res.slope == 0) chi = par.max_chi; // yield is consistently zero if B is very low. Then model slope will be 0. In that case, chi will be NaN. Set chi to max so that we can recover at least some yield later
295 else chi = linreg_predict_inverse(-log(1 - (yield_remainder/bs_remainder)), res);
296 }
297 else if (par.control_model == "linear"){
298 // linear model: y = Bs k X
299 // std::cout << "using linear model" << std::endl;
300 if (yield_remainder >= bs_remainder) chi = par.max_chi;
301 else if (res.slope == 0) chi = par.max_chi;
302 else chi = linreg_predict_inverse((yield_remainder/bs_remainder), res);
303 }
304
305 // if (std::isinf(chi) || std::isnan(chi) || chi > 1e20) throw std::runtime_error("Regressed chi is Inf or NA or extremely large");
306
307 chi = std::clamp(chi, 1e-6, par.max_chi);
308
309}

491double Fleet::effort_constantC(double q, double b, double K){
492 double effort = 0;
493 for (auto& w : window_props_vec){
494 double N0 = w.B_start / K;
495 double C = w.C_rate / K;
496 double M = w.M_fishable;
497 double t = par.window_dt/2;
498
499 double effort_t;
500 double pow_term = (N0+C/M)*exp(-M*t) - C/M;
501 if (C <= 0 || M == 0 || N0 <= 0 || std::isnan(M) || pow_term <= 0) effort_t = 0;
502 else effort_t = C/q/(pow(pow_term, b));
503 effort += effort_t*par.window_dt;
504
505 if (debug || std::isnan(effort_t)) std::cout << "N0: " << N0 << ", C: " << C << ", M: " << M << ", t: " << t << ", effort_t: " << effort_t
506 << ", (N0+C/M)*exp(-M*t): " << (N0+C/M)*exp(-M*t) << ", C/M: " << C/M << std::endl;
507 }
508 return effort;
509}
